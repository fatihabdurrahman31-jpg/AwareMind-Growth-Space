SUPABASE POSTGRESQL + RLS SCHEMA

Skema Database AwareMind Growth Space & Referral AMA7

-- AWAREMIND GROWTH SPACE — SUPABASE E-COMMERCE & AMA7 REFERRAL SCHEMA
CREATE TABLE IF NOT EXISTS public.profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  full_name TEXT NOT NULL,
  plan_tier TEXT DEFAULT 'free' CHECK (plan_tier IN ('free', 'premium')),
  referral_code TEXT UNIQUE NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.referral_directory (
  code TEXT PRIMARY KEY, -- Contoh: AMA7DOP-SHAILA, AMA7END-MUTIARA, AMA7OXY-SORAYA, AMA7SER-FEGI
  ambassador_name TEXT NOT NULL,
  team_name TEXT NOT NULL CHECK (team_name IN ('DOPAMINE', 'ENDORPHIN', 'OXYTOCIN', 'SEROTONIN')),
  discount_percent INT DEFAULT 5,
  is_active BOOLEAN DEFAULT true
);

CREATE TABLE IF NOT EXISTS public.creator_products (
  id TEXT PRIMARY KEY,
  creator_name TEXT NOT NULL,
  space_id INT NOT NULL CHECK (space_id BETWEEN 1 AND 5),
  service_type TEXT NOT NULL CHECK (service_type IN ('toolkit', 'class', 'session')),
  title TEXT NOT NULL,
  price_idr INT NOT NULL,
  cover_image_url TEXT,
  asset_url TEXT NOT NULL,
  preview_text TEXT NOT NULL,
  approval_status TEXT DEFAULT 'pending' CHECK (approval_status IN ('pending', 'approved', 'rejected')),
  sold_verified INT DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  buyer_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  product_id TEXT NOT NULL,
  referral_used TEXT REFERENCES public.referral_directory(code),
  discount_idr INT DEFAULT 0,
  final_paid_idr INT NOT NULL,
  platform_fee_10pct INT GENERATED ALWAYS AS (ROUND(final_paid_idr * 0.10)) STORED,
  creator_payout_90pct INT GENERATED ALWAYS AS (final_paid_idr - ROUND(final_paid_idr * 0.10)) STORED,
  status TEXT DEFAULT 'paid' CHECK (status IN ('paid', 'cancelled', 'refunded')),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.daily_journals (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  space_code TEXT DEFAULT 'SELF_SPACE',
  mood_score INT CHECK (mood_score BETWEEN 1 AND 5),
  journal_text TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE public.daily_journals ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Private user journals" ON public.daily_journals FOR ALL USING (auth.uid() = user_id);
