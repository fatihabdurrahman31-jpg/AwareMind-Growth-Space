SUPABASE POSTGRESQL + RLS SCHEMA

Skema Database E-Commerce AwareMind Growth Space

Salin dan jalankan skrip SQL di bawah ini pada Supabase SQL Editor untuk mengaktifkan tabel akun Freemium, katalog 5 Ruang, transaksi QRIS + referral 5%, penghitung otomatis 100 pembeli, dan jurnal pribadi berproteksi RLS: 

-- AWAREMIND GROWTH SPACE — SUPABASE E-COMMERCE SCHEMA
CREATE TABLE IF NOT EXISTS public.profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  full_name TEXT NOT NULL,
  plan_tier TEXT DEFAULT 'free' CHECK (plan_tier IN ('free', 'premium')),
  referral_code TEXT UNIQUE NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.products (
  id TEXT PRIMARY KEY,
  space_code TEXT NOT NULL, -- SELF_SPACE, PURPOSE, BRAIN_BOOST, FUTURE_READY, CREATE
  service_type TEXT NOT NULL CHECK (service_type IN ('toolkit', 'class', 'session')),
  title TEXT NOT NULL,
  price_idr INT NOT NULL,
  is_priority_mvp BOOLEAN DEFAULT true,
  whatsapp_group_url TEXT,
  sold_verified_count INT DEFAULT 0
);

CREATE TABLE IF NOT EXISTS public.orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  buyer_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  product_id TEXT REFERENCES public.products(id),
  referral_used TEXT,
  discount_idr INT DEFAULT 0,
  final_paid_idr INT NOT NULL,
  status TEXT DEFAULT 'paid' CHECK (status IN ('paid', 'cancelled', 'refunded')),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.daily_journals (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  space_code TEXT DEFAULT 'SELF_SPACE',
  mood_score INT CHECK (mood_score BETWEEN 1 AND 5),
  journal_text TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.daily_journals ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Private user journals" ON public.daily_journals FOR ALL USING (auth.uid() = user_id);
